\section*{Đáp án và lời giải - Đề 02}
\begin{enumerate}
\item Liên tục tại 1: $3=a+b$. Khả vi tại 1: đạo hàm trái bằng $1$, đạo hàm phải bằng $2a$, nên $2a=1$, $a=1/2$. Suy ra $b=3-1/2=5/2$.
\item Theo định nghĩa, với $h(x)=\sqrt{2x+3}$,
\[
h'(x)=\lim_{t\to0}\frac{\sqrt{2x+2t+3}-\sqrt{2x+3}}t.
\]
Nhân liên hợp, được $\displaystyle\lim_{t\to0}\frac{2}{\sqrt{2x+2t+3}+\sqrt{2x+3}}=\frac1{\sqrt{2x+3}}$.
\item $V=\frac43\pi r^3$, nên $dV=4\pi r^2dr=4\pi\cdot9\cdot0.02=0.72\pi$ cm$^3$.
\item Đạo hàm ẩn: $2xy+x^2y'+2yy'=0$, nên $y'=-\frac{2xy}{x^2+2y}$. Tại $(2,1)$, $y'=-4/6=-2/3$. Tiếp tuyến: $y-1=-\frac23(x-2)$.
\item $\displaystyle\lim_{x\to0}\frac{\sin4x}{\tan2x}=\lim_{x\to0}\frac{\sin4x\cos2x}{\sin2x}=2$.
\end{enumerate}
