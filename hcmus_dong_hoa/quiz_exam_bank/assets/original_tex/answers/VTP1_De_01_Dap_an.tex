\section*{Đáp án và lời giải - Đề 01}
\begin{enumerate}
\item Tại $x=0$: $f(0)=1$, $\lim_{x\to0^-}f(x)=1$ và $\lim_{x\to0^+}(ax+1)=1$, nên liên tục với mọi $a$. Tại $x=2$: $\lim_{x\to2^-}f(x)=2a+1$; $f(2)=4-6+b=b-2$. Điều kiện liên tục là $2a+1=b-2$, tức $b=2a+3$. Với $a=2$, $b=7$.
\item Theo định nghĩa,
\[
g'(x)=\lim_{h\to0}\frac{\frac1{x+h+1}-\frac1{x+1}}h
=\lim_{h\to0}\frac{-1}{(x+h+1)(x+1)}=-\frac1{(x+1)^2}.
\]
\item $A=s^2$, nên $dA=2s\,ds=2\cdot4\cdot0.01=0.08\text{ cm}^2$. Sai số diện tích xấp xỉ $0.08$ cm$^2$.
\item Lấy đạo hàm hai vế: $2x+y+xy'+2yy'=0$. Do đó $y'=-(2x+y)/(x+2y)$. Tại $(1,2)$, $y'=-4/5$. Tiếp tuyến: $y-2=-\frac45(x-1)$.
\item Dạng $0/0$. Dùng L'Hospital hai lần:
\[
\lim_{x\to0}\frac{e^{3x}-1-3x}{x^2}=\lim_{x\to0}\frac{3e^{3x}-3}{2x}=\lim_{x\to0}\frac{9e^{3x}}2=\frac92.
\]
\end{enumerate}
