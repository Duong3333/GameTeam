\section*{Đáp án và lời giải - Đề 04}
\begin{enumerate}
\item Với $x\ne2$, $\frac{x^2-4}{x-2}=x+2$. Giới hạn tại 2 bằng 4, nên $c=4$. Khi đó hàm trùng với $x+2$ trên toàn miền, do đó khả vi và $f'(2)=1$.
\item $g'(x)=\lim_{h\to0}\frac{1/(2x+2h+1)-1/(2x+1)}h=\lim_{h\to0}\frac{-2}{(2x+2h+1)(2x+1)}=-\frac2{(2x+1)^2}$.
\item $dC=2\pi\,dr=2\pi(-0.03)=-0.06\pi$ cm. Chu vi giảm xấp xỉ $0.06\pi$ cm.
\item Lấy đạo hàm: $y+xy'+2yy'=0$, do đó $y'=-y/(x+2y)$. Tại $(2,2)$, hệ số góc $-2/6=-1/3$. Tiếp tuyến: $y-2=-\frac13(x-2)$.
\item Dùng giới hạn chuẩn $1-\cos u\sim u^2/2$: kết quả $25/2$. Có thể dùng L'Hospital hai lần để thu được cùng kết quả.
\end{enumerate}
