\section*{Đáp án và lời giải - Đề 03}
\begin{enumerate}
\item Tại $x=2$, giới hạn trái bằng $2(2)+1=5$, giá trị/giới hạn phải bằng $4+2m+1=5+2m$. Liên tục khi $m=0$. Khi đó đạo hàm trái bằng $2$, đạo hàm phải bằng $2x+m$ tại 2 bằng $4$, nên không khả vi tại 2.
\item $f'(x)=\lim_{h\to0}\frac{(x+h)^2-3(x+h)-(x^2-3x)}h=\lim_{h\to0}(2x+h-3)=2x-3$.
\item $f(4)=3$, $f'(x)=\frac1{2\sqrt{x+5}}$, $f'(4)=1/6$. Vì vậy $L(x)=3+\frac16(x-4)$. $\sqrt{9.1}=f(4.1)\approx L(4.1)=3+0.1/6=3.016\overline6$.
\item $2x+2yy'=0$, nên $y'=-x/y$. Tại $(3,4)$, hệ số góc $-3/4$. Tiếp tuyến: $y-4=-\frac34(x-3)$.
\item Dạng $0/0$, L'Hospital hai lần:
\[
\lim_{x\to0}\frac{\ln(1+x)-x}{x^2}=\lim_{x\to0}\frac{1/(1+x)-1}{2x}=\lim_{x\to0}\frac{-1/(1+x)^2}{2}=-\frac12.
\]
\end{enumerate}
