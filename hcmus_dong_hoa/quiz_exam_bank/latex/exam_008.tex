\studentline
\begin{notebox}\textbf{Lưu ý:} Câu xấp xỉ tuyến tính yêu cầu nêu rõ điểm khai triển và công thức sử dụng.\end{notebox}
\begin{enumerate}
\item Xét hàm số $f(x)=\begin{cases}2x+1,&x<2,\\x^2+mx+1,&x\ge2.\end{cases}$ Tìm $m$ để $f$ liên tục tại $x=2$. Khi đó, hàm số có khả vi tại $x=2$ không? \points{2}
\item Dùng định nghĩa đạo hàm để tính đạo hàm của $f(x)=x^2-3x$. \points{2}
\item Tìm hàm xấp xỉ tuyến tính của $f(x)=\sqrt{x+5}$ tại $x_0=4$, rồi dùng nó ước lượng $\sqrt{9.1}$. \points{2}
\item Với đường tròn $x^2+y^2=25$, tìm phương trình tiếp tuyến tại $(3,4)$. \points{2}
\item Tính $\displaystyle\lim_{x\to0}\frac{\ln(1+x)-x}{x^2}$. \points{2}
\end{enumerate}
