\studentline
\begin{notebox}\textbf{Lưu ý:} Các câu hỏi kiểm tra giới hạn, đạo hàm theo định nghĩa, vi phân và đạo hàm hàm ẩn.\end{notebox}
\begin{enumerate}
\item Cho
\[
f(x)=\begin{cases}x+2,&x<1,\\ ax^2+b,&x\ge 1.\end{cases}
\]
Tìm $a,b$ để $f$ liên tục và khả vi tại $x=1$. \points{2}
\item Dùng định nghĩa đạo hàm để tìm đạo hàm của $g(x)=\sqrt{2x+3}$ trên miền xác định. \points{2}
\item Một quả cầu có bán kính $r=3$ cm. Bán kính tăng thêm một lượng nhỏ $dr=0.02$ cm. Dùng vi phân để ước tính độ tăng thể tích. \points{1.5}
\item Đường cong $x^2y+y^2=6$ đi qua điểm $(2,1)$. Tìm hệ số góc và phương trình tiếp tuyến tại điểm đó. \points{2.5}
\item Tính $\displaystyle\lim_{x\to0}\frac{\sin(4x)}{\tan(2x)}$. \points{2}
\end{enumerate}
